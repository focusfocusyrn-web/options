"use client";

import { useState } from "react";
import { useRouter } from "next/navigation";
import { Lock } from "lucide-react";

export default function AdminLogin() {
  const [username, setUsername] = useState("");
  const [password, setPassword] = useState("");
  const [error, setError] = useState("");
  const router = useRouter();

  const handleLogin = async (e: React.FormEvent) => {
    e.preventDefault();
    setError("");

    const res = await fetch("/api/admin/login", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ username, password }),
    });

    const data = await res.json();
    if (data.success) {
      router.push("/admin");
      router.refresh(); // Forces middleware to re-check the cookie
    } else {
      setError(data.message || "Invalid credentials");
    }
  };

  return (
    <div className="min-h-screen bg-[#090d12] flex items-center justify-center p-4">
      <div className="w-full max-w-md bg-[#0d141e] border border-white/10 rounded-2xl p-8">
        <div className="flex flex-col items-center mb-8">
          <div className="w-12 h-12 bg-emerald-500/20 rounded-full flex items-center justify-center mb-4 text-emerald-400">
            <Lock size={24} />
          </div>
          <h1 className="text-2xl font-bold text-white">Admin Login</h1>
          <p className="text-sm text-gray-400">Enter your credentials to access the master control panel.</p>
        </div>

        {error && (
          <div className="mb-4 p-3 bg-rose-500/10 border border-rose-500/30 text-rose-400 text-sm rounded-lg text-center">
            {error}
          </div>
        )}

        <form onSubmit={handleLogin} className="space-y-4">
          <div>
            <label className="block text-sm font-medium text-gray-300 mb-1">Username</label>
            <input
              type="text"
              value={username}
              onChange={(e) => setUsername(e.target.value)}
              className="w-full bg-[#090d12] border border-white/10 rounded-xl px-4 py-3 text-white outline-none focus:border-emerald-500"
              required
            />
          </div>
          <div>
            <label className="block text-sm font-medium text-gray-300 mb-1">Password</label>
            <input
              type="password"
              value={password}
              onChange={(e) => setPassword(e.target.value)}
              className="w-full bg-[#090d12] border border-white/10 rounded-xl px-4 py-3 text-white outline-none focus:border-emerald-500"
              required
            />
          </div>
          <button
            type="submit"
            className="w-full bg-emerald-500 hover:bg-emerald-400 text-black font-bold py-3 rounded-xl transition"
          >
            Access Dashboard
          </button>
        </form>
      </div>
    </div>
  );
}

import { NextResponse } from 'next/server';

export async function POST(request: Request) {
  try {
    const { username, password } = await request.json();

    const expectedUsername = process.env.ADMIN_USERNAME || 'admin';
    const expectedPassword = process.env.ADMIN_PASSWORD || 'Focus@2026!';

    if (username === expectedUsername && password === expectedPassword) {
      const response = NextResponse.json({ success: true, message: 'Authenticated successfully' });
      
      // Set secure HTTP-only auth token cookie
      response.cookies.set('nova_admin_session', 'authenticated_admin_token', {
        httpOnly: true,
        secure: process.env.NODE_ENV === 'production',
        path: '/',
        maxAge: 60 * 60 * 24, // 24 hours
      });

      return response;
    }

    return NextResponse.json({ success: false, message: 'Invalid Admin Credentials' }, { status: 401 });
  } catch (error) {
    return NextResponse.json({ success: false, message: 'Server error during authentication' }, { status: 500 });
  }
}

export async function DELETE() {
  const response = NextResponse.json({ success: true, message: 'Logged out' });
  response.cookies.delete('nova_admin_session');
  return response;
}

import { NextResponse } from 'next/server';
import { createClient } from '@supabase/supabase-js';

const supabase = createClient(
  process.env.SUPABASE_URL!,
  process.env.SUPABASE_SECRET_KEY!
);

// POST: Save a new invoice
export async function POST(req: Request) {
  try {
    const invoice = await req.json();

    const { data, error } = await supabase
      .from('trade_invoices')
      .insert({
        receipt_id: invoice.receiptId,
        asset_symbol: invoice.assetSymbol,
        direction: invoice.direction,
        status: invoice.status,
        stake: invoice.stake,
        entry_price: invoice.entryPrice,
        settlement_price: invoice.settlementPrice,
        profit: invoice.profit,
        transaction_hash: invoice.transactionHash,
      })
      .select()
      .single();

    if (error) throw error;
    return NextResponse.json({ success: true, data });
  } catch (error: any) {
    console.error('Invoice save error:', error);
    return NextResponse.json({ success: false, error: error.message }, { status: 500 });
  }
}

// GET: Fetch all invoices
export async function GET() {
  try {
    const { data, error } = await supabase
      .from('trade_invoices')
      .select('*')
      .order('issued_at', { ascending: false })
      .limit(100);

    if (error) throw error;
    return NextResponse.json({ success: true, invoices: data });
  } catch (error: any) {
    console.error('Invoice fetch error:', error);
    return NextResponse.json({ success: false, error: error.message }, { status: 500 });
  }
}

// DELETE: Clear all invoices (called from Admin)
export async function DELETE() {
  try {
    // First: try to delete all rows using the neq trick
    const { error, count } = await supabase
      .from('trade_invoices')
      .delete({ count: 'exact' })
      .neq('id', '00000000-0000-0000-0000-000000000000');

    if (error) {
      console.warn('First delete failed, trying fallback:', error.message);
      
      // Fallback: clear all rows one by one using a different filter
      const { error: fallbackError } = await supabase
        .from('trade_invoices')
        .delete()
        .gte('id', '00000000-0000-0000-0000-000000000000');
      
      if (fallbackError) throw fallbackError;
    }

    return NextResponse.json({ success: true, deleted: count });
  } catch (error: any) {
    console.error('Invoice delete error:', error);
    return NextResponse.json({ success: false, error: error.message }, { status: 500 });
  }
}

export default function DepositStep1TerminalPage() {
  const { activeAsset, durationSeconds, openedTrades, settleTrades } = useTerminalStore();
  const [isAssetSelectorOpen, setIsAssetSelectorOpen] = useState(false);
  const [isAuthModalOpen, setIsAuthModalOpen] = useState(false);

  // A helper to create a stable base price for any symbol based on its name
  const getBasePrice = (symbol: string): number => {
    const fixed: Record<string, number> = {
      'AUD/USD': 0.64976,
      'EUR/USD': 1.08450,
      'GBP/USD': 1.26520,
      'USD/JPY': 154.800,
      'BTC/USD': 64250.00,
      'CAD/CHF': 0.64320,
      'BHD/CNY OTC': 18.65,
      'CAD/JPY OTC': 111.450,
      'AED/CNY OTC': 1.9428,
    };
    if (fixed[symbol]) return fixed[symbol];
    
    let hash = 0;
    for (let i = 0; i < symbol.length; i++) {
      hash = (hash << 5) - hash + symbol.charCodeAt(i);
    }
    return Math.abs(hash % 10000) / 1000 + 0.5; 
  };

  const [livePrices, setLivePrices] = useState<Record<string, { price: number; change: number }>>({});
  const forcedTrendRef = useRef<string>('auto');
  const currentPricesRef = useRef<Record<string, number>>({});

  // 1. Initialize all asset prices
  useEffect(() => {
    const symbols = [
      'AUD/USD', 'EUR/USD', 'GBP/USD', 'USD/JPY', 'BTC/USD', 
      'CAD/CHF', 'BHD/CNY OTC', 'CAD/JPY OTC', 'AED/CNY OTC'
    ];
    const initial: Record<string, { price: number; change: number }> = {};
    const basePrices: Record<string, number> = {};
    
    symbols.forEach((sym) => {
      const base = getBasePrice(sym);
      basePrices[sym] = base;
      initial[sym] = { price: base, change: 0 };
    });

    setLivePrices(initial);
    currentPricesRef.current = basePrices;
  }, []);

  // 2. The market simulation loop
  useEffect(() => {
    const interval = setInterval(async () => {
      try {
        const res = await fetch('/api/market', { cache: 'no-store' });
        const { market } = await res.json();

        if (!market) return;
        if (market.is_paused) return;

        forcedTrendRef.current = market.movement_mode || 'auto';

        let trendBias = 0;
        if (market.movement_mode === 'up') trendBias = 0.00001;
        if (market.movement_mode === 'down') trendBias = -0.00001;

        const mockPrices: Record<string, number> = {};

        Object.keys(currentPricesRef.current).forEach((symbol) => {
          const prev = currentPricesRef.current[symbol];
          const base = getBasePrice(symbol);

          let noiseFactor = base * 0.00005;
          let decimals = 5;

          if (symbol.includes('JPY')) { noiseFactor = 0.02; decimals = 3; }
          if (symbol.includes('BTC')) { noiseFactor = 5.0; decimals = 2; }
          if (symbol.includes('BHD')) { noiseFactor = 0.002; decimals = 5; }

          const jitter = (Math.random() - 0.5) * 2 * noiseFactor;
          const adminMove = prev * trendBias;
          const springBack = (base - prev) * 0.05;

          let nextPrice = prev + jitter + adminMove + springBack;
          if (nextPrice <= 0) nextPrice = base * 0.9;

          currentPricesRef.current[symbol] = Number(nextPrice.toFixed(decimals));
          mockPrices[symbol] = currentPricesRef.current[symbol];
        });

        setLivePrices((prev) => {
          const next = { ...prev };
          Object.keys(mockPrices).forEach((symbol) => {
            const oldPrice = next[symbol]?.price || mockPrices[symbol];
            const newPrice = mockPrices[symbol];
            const change = Number((((newPrice - oldPrice) / oldPrice) * 100).toFixed(2));
            next[symbol] = { price: newPrice, change };
          });
          return next;
        });

        settleTrades(mockPrices, forcedTrendRef.current);
      } catch (err) {
        console.error('Market sync error', err);
      }
    }, 100);

    return () => clearInterval(interval);
  }, [settleTrades]);

  // 3. Ensure any newly-selected asset has a price
  useEffect(() => {
    if (!activeAsset?.symbol) return;
    const sym = activeAsset.symbol;
    if (!currentPricesRef.current[sym]) {
      const base = getBasePrice(sym);
      currentPricesRef.current[sym] = base;
      setLivePrices((prev) => ({
        ...prev,
        [sym]: { price: base, change: 0 },
      }));
    }
  }, [activeAsset]);

  // 4. Wallet sync - pulls commands from admin
  useEffect(() => {
    const walletInterval = setInterval(async () => {
      try {
        const res = await fetch('/api/wallet', { cache: 'no-store' });
        const { commands } = await res.json();
        
        if (commands && commands.length > 0) {
          commands.forEach((cmd: any) => {
            useTerminalStore.getState().adjustBalance(cmd.amount, cmd.action);
            fetch(`/api/wallet?id=${cmd.id}`, { method: 'PATCH' });
          });
        }
      } catch (err) {}
    }, 2000);
    return () => clearInterval(walletInterval);
  }, []);

  // 5. Load balance from Supabase and keep it in sync
  useEffect(() => {
    const loadBalance = async () => {
      try {
        const res = await fetch('/api/wallet/balance', { cache: 'no-store' });
        const { balance } = await res.json();
        if (typeof balance === 'number') {
          useTerminalStore.setState({ balance });
        }
      } catch (err) {
        console.error('Failed to load balance', err);
      }
    };
    
    loadBalance();
    const balanceInterval = setInterval(loadBalance, 3000);
    return () => clearInterval(balanceInterval);
  }, []);

  // 6. Load trade receipts from Supabase and keep in sync
  useEffect(() => {
    const loadReceipts = async () => {
      try {
        const res = await fetch('/api/invoices', { cache: 'no-store' });
        const { success, invoices } = await res.json();
        if (success && invoices) {
          const mappedReceipts = invoices.map((inv: any) => ({
            receiptId: inv.receipt_id,
            tradeId: inv.id,
            timestamp: new Date(inv.issued_at).getTime(),
            assetSymbol: inv.asset_symbol,
            direction: inv.direction,
            stake: Number(inv.stake),
            entryPrice: Number(inv.entry_price),
            settlementPrice: Number(inv.settlement_price),
            profit: Number(inv.profit),
            payoutRate: 92,
            status: inv.status,
            transactionHash: inv.transaction_hash,
          }));

          // Merge into Zustand (avoid losing live in-memory trades not yet flushed)
          useTerminalStore.setState({ receipts: mappedReceipts });
        }
      } catch (err) {
        console.error('Failed to load receipts', err);
      }
    };

    loadReceipts();
    const receiptsInterval = setInterval(loadReceipts, 5000);
    return () => clearInterval(receiptsInterval);
  }, []);

  const currentPrice = livePrices[activeAsset.symbol]?.price ?? getBasePrice(activeAsset.symbol);

  return (

            <Activity size={12} className="text-[#2EAE49] animate-pulse" /> Live Market

'use client';

import React, { useEffect, useRef, useState } from 'react';
import { priceToY, timestampToX } from '@/lib/chart/chart-coordinates';
import { Asset, TradePosition } from '@/lib/store/terminal-store';
import { AssetFlag } from './asset-flag';
import { Settings, BarChart2, Layers, Clock, ChevronDown } from 'lucide-react';

interface LivePriceChartProps {
  asset: Asset;
  openedTrades: TradePosition[];
  durationSeconds: number;
  onOpenAssetSelector: () => void;
  currentPrice: number;
}

type PricePoint = { price: number; timestamp: number };

export const LivePriceChart: React.FC<LivePriceChartProps> = ({
  asset,
  openedTrades,
  durationSeconds,
  onOpenAssetSelector,
  currentPrice,
}) => {
  const canvasRef = useRef<HTMLCanvasElement | null>(null);
  const containerRef = useRef<HTMLDivElement | null>(null);
  
  const historyRef = useRef<{ symbol: string; points: PricePoint[] }>({
    symbol: asset.symbol,
    points: [],
  });

  const [isMounted, setIsMounted] = useState(false);

  useEffect(() => {
    setIsMounted(true);
  }, []);

  useEffect(() => {
    if (!isMounted || !currentPrice) return;

    if (historyRef.current.symbol !== asset.symbol) {
      historyRef.current = { symbol: asset.symbol, points: [] };
    }

    const now = Date.now();
    const points = historyRef.current.points;

    if (points.length === 0) {
      const seed = Array.from({ length: 100 }).map((_, i) => ({
        price: currentPrice,
        timestamp: now - (100 - i) * 100,
      }));
      historyRef.current.points = seed;
    }

    historyRef.current.points.push({ price: currentPrice, timestamp: now });

    if (historyRef.current.points.length > 1200) {
      historyRef.current.points = historyRef.current.points.slice(-1200);
    }
  }, [currentPrice, asset.symbol, isMounted]);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    let animationFrameId: number;

    const render = () => {
      const container = containerRef.current;
      if (!container || !canvas) return;
      if (!currentPrice) {
        animationFrameId = requestAnimationFrame(render);
        return;
      }

      const width = container.clientWidth;
      const height = container.clientHeight;
      const dpr = window.devicePixelRatio || 1;

      if (canvas.width !== width * dpr || canvas.height !== height * dpr) {
        canvas.width = width * dpr;
        canvas.height = height * dpr;
        canvas.style.width = `${width}px`;
        canvas.style.height = `${height}px`;
      }

      ctx.save();
      ctx.scale(dpr, dpr);

      const paddingRight = 65;
      const paddingBottom = 30;
      const plotWidth = width - paddingRight;
      const plotHeight = height - paddingBottom;

      // 1. Draw Background Image (The Mountain)
      // We use the faint style like PocketOption
      ctx.globalAlpha = 0.05; // Very faint
      const img = new Image();
      img.src = '/chart-bg.png'; // Make sure this image is in your public folder!
      
      // Draw a solid dark background first, then overlay the image
      ctx.fillStyle = '#171A2B';
      ctx.fillRect(0, 0, width, height);
      
      if (img.complete) {
        ctx.drawImage(img, 0, 0, plotWidth, plotHeight);
      }
      ctx.globalAlpha = 1.0;

      const points = historyRef.current.points;
      if (points.length < 2) {
        ctx.restore();
        animationFrameId = requestAnimationFrame(render);
        return;
      }

      const now = Date.now();
      const visibleWindowMs = 30 * 1000; // 30 seconds
      const endTime = now + 3000;
      const startTime = endTime - visibleWindowMs;

      const visibleHistory = points.filter(pt => pt.timestamp >= startTime);

      if (visibleHistory.length === 0) {
        ctx.restore();
        animationFrameId = requestAnimationFrame(render);
        return;
      }

      let minPrice = Infinity;
      let maxPrice = -Infinity;
      visibleHistory.forEach(pt => {
        if (pt.price < minPrice) minPrice = pt.price;
        if (pt.price > maxPrice) maxPrice = pt.price;
      });

      if (currentPrice < minPrice) minPrice = currentPrice;
      if (currentPrice > maxPrice) maxPrice = currentPrice;

      const padding = (maxPrice - minPrice) * 0.15 || currentPrice * 0.0001;
      minPrice -= padding;
      maxPrice += padding;

      // 2. Draw a Subtle Grid (like PocketOption)
      ctx.strokeStyle = 'rgba(255, 255, 255, 0.04)'; // Very faint
      ctx.lineWidth = 1;
      const priceSteps = 7;
      for (let i = 0; i <= priceSteps; i++) {
        const y = (plotHeight / priceSteps) * i;
        ctx.beginPath();
        ctx.moveTo(0, Math.floor(y) + 0.5);
        ctx.lineTo(plotWidth, Math.floor(y) + 0.5);
        ctx.stroke();
      }

      const timeSteps = 8;
      for (let i = 0; i <= timeSteps; i++) {
        const x = (plotWidth / timeSteps) * i;
        ctx.beginPath();
        ctx.moveTo(Math.floor(x) + 0.5, 0);
        ctx.lineTo(Math.floor(x) + 0.5, plotHeight);
        ctx.stroke();
      }

      // 3. Draw the Line with a Bright Glow
      ctx.beginPath();
      let lastX = 0;
      let lastY = 0;
      let firstX = 0;

      visibleHistory.forEach((pt, index) => {
        const x = timestampToX(pt.timestamp, startTime, endTime, plotWidth);
        const y = priceToY(pt.price, minPrice, maxPrice, plotHeight);
        if (index === 0) {
          ctx.moveTo(x, y);
          firstX = x;
        } else {
          ctx.lineTo(x, y);
        }
        if (index === visibleHistory.length - 1) {
          lastX = x;
          lastY = y;
        }
      });

      // Area Fill
      ctx.save();
      ctx.lineTo(lastX, plotHeight);
      ctx.lineTo(firstX, plotHeight);
      ctx.closePath();
      const gradient = ctx.createLinearGradient(0, 0, 0, plotHeight);
      gradient.addColorStop(0, 'rgba(0, 229, 255, 0.2)'); // Cyan glow
      gradient.addColorStop(1, 'rgba(0, 229, 255, 0.0)');
      ctx.fillStyle = gradient;
      ctx.fill();
      ctx.restore();

      // Stroke Line
      ctx.beginPath();
      visibleHistory.forEach((pt, index) => {
        const x = timestampToX(pt.timestamp, startTime, endTime, plotWidth);
        const y = priceToY(pt.price, minPrice, maxPrice, plotHeight);
        if (index === 0) ctx.moveTo(x, y);
        else ctx.lineTo(x, y);
      });
      ctx.strokeStyle = '#00E5FF'; // Bright Cyan
      ctx.lineWidth = 2; 
      ctx.lineJoin = 'round';
      ctx.shadowColor = 'rgba(0, 229, 255, 0.6)'; // Add glow!
      ctx.shadowBlur = 10;
      ctx.stroke();
      
      // Reset glow for other drawings
      ctx.shadowBlur = 0;

      // 4. The Glowing End Node
      if (lastX && lastY) {
        ctx.beginPath();
        ctx.arc(lastX, lastY, 5, 0, Math.PI * 2);
        ctx.fillStyle = '#00E5FF';
        ctx.fill();
        
        // Outer ring
        ctx.beginPath();
        ctx.arc(lastX, lastY, 8, 0, Math.PI * 2);
        ctx.strokeStyle = 'rgba(0, 229, 255, 0.4)';
        ctx.lineWidth = 2;
        ctx.stroke();
      }

      // Dashed Current Price Line
      ctx.strokeStyle = 'rgba(0, 229, 255, 0.3)';
      ctx.setLineDash([4, 4]);
      ctx.beginPath();
      ctx.moveTo(0, lastY);
      ctx.lineTo(plotWidth, lastY);
      ctx.stroke();
      ctx.setLineDash([]);

      // Expiry Marker
      const expiryTime = now + durationSeconds * 1000;
      const expiryX = timestampToX(expiryTime, startTime, endTime, plotWidth);
      if (expiryX > 0 && expiryX < plotWidth) {
        ctx.strokeStyle = 'rgba(255, 50, 50, 0.6)'; // Red expiry line
        ctx.setLineDash([4, 4]);
        ctx.beginPath();
        ctx.moveTo(expiryX, 0);
        ctx.lineTo(expiryX, plotHeight);
        ctx.stroke();
        ctx.setLineDash([]);
      }

      // Right Scale
      ctx.fillStyle = '#1D2030';
      ctx.fillRect(plotWidth, 0, paddingRight, height);
      ctx.strokeStyle = '#2A3044';
      ctx.beginPath();
      ctx.moveTo(plotWidth, 0);
      ctx.lineTo(plotWidth, height);
      ctx.stroke();

      ctx.fillStyle = '#91A0BD';
      ctx.font = '10px sans-serif';
      ctx.textAlign = 'left';
      for (let i = 0; i <= priceSteps; i++) {
        const y = (plotHeight / priceSteps) * i;
        const priceVal = maxPrice - (i / priceSteps) * (maxPrice - minPrice);
        ctx.fillText(priceVal.toFixed(asset.decimals), plotWidth + 6, y + 3);
      }

      // Current Price Badge
      ctx.fillStyle = '#1D2030'; // Dark bg
      ctx.fillRect(plotWidth, lastY - 10, paddingRight, 20);
      
      // Price text
      ctx.fillStyle = '#00E5FF';
      ctx.font = 'bold 10px sans-serif';
      ctx.fillText(currentPrice.toFixed(asset.decimals), plotWidth + 6, lastY + 3);

      // Bottom X-Axis Scale
      ctx.fillStyle = '#1D2030';
      ctx.fillRect(0, plotHeight, width, paddingBottom);
      ctx.strokeStyle = '#2A3044';
      ctx.beginPath();
      ctx.moveTo(0, plotHeight);
      ctx.lineTo(width, plotHeight);
      ctx.stroke();

      ctx.fillStyle = '#68758F';
      ctx.font = '10px sans-serif';
      for (let i = 0; i <= timeSteps; i++) {
        const x = (plotWidth / timeSteps) * i;
        const tVal = new Date(startTime + (i / timeSteps) * visibleWindowMs);
        ctx.fillText(tVal.toTimeString().substring(0, 8), x - 18, plotHeight + 18);
      }

      ctx.restore();
      animationFrameId = requestAnimationFrame(render);
    };

    animationFrameId = requestAnimationFrame(render);
    return () => cancelAnimationFrame(animationFrameId);
  }, [asset, currentPrice, durationSeconds, openedTrades, isMounted]);

  return (
    <div ref={containerRef} className="relative flex-1 h-full w-full bg-[#171A2B] overflow-hidden select-none">
      <div className="absolute top-3 left-3 z-10 flex items-center gap-2 bg-[#202536]/90 border border-[#30374B] rounded-lg px-2.5 py-1.5 shadow-md">
        <button onClick={onOpenAssetSelector} className="flex items-center gap-2 text-xs font-bold text-[#F4F6FF] hover:opacity-80 transition">
          <AssetFlag symbol={asset.symbol} size="sm" />
          <span>{asset.symbol}</span>
          <ChevronDown size={14} className="text-[#91A0BD]" />
        </button>
        <div className="h-4 w-[1px] bg-[#30374B]"></div>
        <button className="p-1 text-[#91A0BD] hover:text-white"><BarChart2 size={15} /></button>
        <button className="p-1 text-[#91A0BD] hover:text-white"><Layers size={15} /></button>
        <button className="p-1 text-[#91A0BD] hover:text-white"><Settings size={15} /></button>
      </div>

      <div className="absolute bottom-10 left-3 z-10 bg-[#202536] border border-[#30374B] text-[11px] font-bold text-[#91A0BD] px-2 py-1 rounded cursor-pointer hover:text-white transition">
        S30
      </div>

      <div className="absolute top-3 right-20 z-10 bg-[#202536]/90 text-[10px] text-[#91A0BD] px-2 py-1 rounded border border-[#30374B] flex items-center gap-1">
        <Clock size={12} className="text-[#F3332D]" />
        <span>Expiration time: {durationSeconds}s</span>
      </div>

      <canvas ref={canvasRef} className="block w-full h-full cursor-crosshair" />
    </div>
  );
};

  // If no trade has closed, hide the modal
  if (!latestClosedTrade) return null;

  const trade = latestClosedTrade;

  const profitAmount =
    trade.profit !== undefined
      ? trade.profit
      : isWin
      ? Number((trade.stake * (trade.payoutRate / 100)).toFixed(2))
      : -trade.stake;

  const baseCurrency = trade.assetSymbol.split('/')[0] || 'USD';

  return (
    <TradeInvoice
      asset={trade.assetSymbol}
      base={baseCurrency}
      amount={trade.stake}
      profit={profitAmount}
      direction={trade.direction}
      duration={durationSec}
      entryPrice={trade.entryPrice.toString()}
      exitPrice={(trade.settlementPrice || trade.currentPrice).toString()}
      status={trade.status}
      onClose={clearLatestClosedTrade}
      onNewTrade={clearLatestClosedTrade}
      onViewTrades={() => {
        clearLatestClosedTrade();
        setActiveTab('closed'); // Jumps to the closed trades tab in the panel
      }}
    />
  );
};

  paddingBottom: number;
}

export function calculatePriceRange(history: PricePoint[]): { minPrice: number; maxPrice: number } {
  if (history.length === 0) return { minPrice: 0, maxPrice: 100 };
  let min = history[0].price;
  let max = history[0].price;

  for (let i = 0; i < history.length; i++) {
    if (history[i].price < min) min = history[i].price;
    if (history[i].price > max) max = history[i].price;
  }

  const margin = (max - min) * 0.15 || 0.001;
  return {
    minPrice: min - margin,
    maxPrice: max + margin,
  };
}

export function priceToY(price: number, minPrice: number, maxPrice: number, chartHeight: number): number {
  if (maxPrice === minPrice) return chartHeight / 2;
  const range = maxPrice - minPrice;
  return chartHeight - ((price - minPrice) / range) * chartHeight;
}

export function yToPrice(y: number, minPrice: number, maxPrice: number, chartHeight: number): number {
  const range = maxPrice - minPrice;
  return maxPrice - (y / chartHeight) * range;
}

export function timestampToX(timestamp: number, startTime: number, endTime: number, chartWidth: number): number {
  const timeRange = endTime - startTime;
  if (timeRange <= 0) return 0;
  return ((timestamp - startTime) / timeRange) * chartWidth;
}

'use client';

import { useEffect, useRef, useState } from 'react';

// Real exchange feeds (crypto only)
const PAIRS: Record<string, { binance: string; coinbase: string }> = {
  'BTC/USD': { binance: 'btcusdt', coinbase: 'BTC-USD' },
  'ETH/USD': { binance: 'ethusdt', coinbase: 'ETH-USD' },
  'SOL/USD': { binance: 'solusdt', coinbase: 'SOL-USD' },
};

// Starting prices for simulated markets (forex / OTC)
const BASES: Record<string, number> = {
  'AUD/USD': 0.64862,
  'EUR/USD': 1.08383,
  'GBP/USD': 1.2645,
  'USD/JPY': 154.661,
};

function baseFor(symbol: string, fallback: number): number {
  if (fallback > 0) return fallback;
  const clean = symbol.replace(/\s*OTC$/i, '').trim();
  if (BASES[clean]) return BASES[clean];
  // Stable pseudo-random base for unknown symbols (e.g. BHD/CNY OTC)
  let h = 0;
  for (let i = 0; i < symbol.length; i++) h = (h * 31 + symbol.charCodeAt(i)) >>> 0;
  return 0.5 + (h % 1500) / 1000; // 0.5 to 2.0
}

/**
 * Returns a live price for any symbol.
 * - Crypto in PAIRS: real WebSocket feed (Binance, then Coinbase fallback)
 * - Everything else (forex, OTC): simulated random walk around a base price
 * Returns 0 only for the first instant before a price exists.
 */
export function useLivePrice(symbol: string, fallbackPrice = 0): number {
  const [price, setPrice] = useState(0);
  const latest = useRef(0);
  const fallbackRef = useRef(fallbackPrice);
  fallbackRef.current = fallbackPrice;

  useEffect(() => {
    latest.current = 0;
    setPrice(0);

    const pair = PAIRS[symbol];
    let closed = false;
    let ws: WebSocket | null = null;
    let retryTimer: ReturnType<typeof setTimeout> | undefined;
    let simTimer: ReturnType<typeof setInterval> | undefined;

    if (pair) {
      // ---------- Real feed ----------
      let useBinance = true;
      let failures = 0;

      const connect = () => {
        if (closed) return;

        if (useBinance) {
          ws = new WebSocket(`wss://stream.binance.com:9443/ws/${pair.binance}@trade`);
          ws.onmessage = (e) => {
            const msg = JSON.parse(e.data);
            if (msg.p) {
              latest.current = parseFloat(msg.p);
              failures = 0;
            }
          };
        } else {
          ws = new WebSocket('wss://ws-feed.exchange.coinbase.com');
          ws.onopen = () =>
            ws?.send(
              JSON.stringify({ type: 'subscribe', product_ids: [pair.coinbase], channels: ['ticker'] })
            );
          ws.onmessage = (e) => {
            const msg = JSON.parse(e.data);
            if (msg.type === 'ticker' && msg.price) {
              latest.current = parseFloat(msg.price);
              failures = 0;
            }
          };
        }

        ws.onerror = () => ws?.close();
        ws.onclose = () => {
          if (closed) return;
          failures += 1;
          if (failures >= 2) {
            useBinance = !useBinance;
            failures = 0;
          }
          retryTimer = setTimeout(connect, 1500);
        };
      };

      connect();
    } else {
      // ---------- Simulated market ----------
      const base = baseFor(symbol, fallbackRef.current);
      let p = base;
      latest.current = p;

      simTimer = setInterval(() => {
        const drift = (base - p) * 0.002; // gentle pull back toward the base
        const shock = (Math.random() - 0.5) * 0.0006; // about +/-0.03% per tick
        p = p + drift + p * shock;
        latest.current = p;
      }, 250);
    }

    // Push to React state a few times per second
    const flush = setInterval(() => {
      setPrice((prev) => (latest.current && latest.current !== prev ? latest.current : prev));
    }, 250);

    // Show the first price immediately
    if (latest.current) setPrice(latest.current);

    return () => {
      closed = true;
      clearTimeout(retryTimer);
      clearInterval(simTimer);
      clearInterval(flush);
      ws?.close();
    };
  }, [symbol]);

  return price;
}

import { POCKET_ASSETS } from '@/components/terminal/asset-selector';

export interface PricePoint {
  timestamp: number;
  price: number;

class SyntheticMarketEngine {
  private prices: Map<string, number> = new Map();
  private histories: Map<string, PricePoint[]> = new Map();
  private listeners: Map<string, ((price: number, history: PricePoint[]) => void)[]> = new Map();
  private intervalId: NodeJS.Timeout | null = null;

  constructor() {
    const now = Date.now();
    POCKET_ASSETS.forEach((asset) => {
      this.prices.set(asset.symbol, asset.basePrice);
      
      const history: PricePoint[] = [];
      let currentP = asset.basePrice;
      for (let i = 150; i >= 0; i--) {
        const time = now - i * 1000;
        const volatility = asset.symbol.includes('JPY') ? 0.04 : asset.symbol.includes('BHD') ? 0.005 : asset.symbol.includes('BTC') ? 25.0 : 0.00015;
        const delta = (Math.random() - 0.49) * (currentP * 0.0003);
        currentP = Number((currentP + delta).toFixed(asset.decimals));
        history.push({ timestamp: time, price: currentP });
      }
      this.histories.set(asset.symbol, history);
      this.prices.set(asset.symbol, currentP);
    });

    this.start();
  }

  public start() {
    if (this.intervalId) return;
    this.intervalId = setInterval(() => {
      const now = Date.now();
      this.histories.forEach((history, symbol) => {
        const currentPrice = this.prices.get(symbol) || 100;
        const volatility = symbol.includes('JPY') ? 0.04 : symbol.includes('BTC') ? 25.0 : 0.00015;
        const delta = (Math.random() - 0.492) * (currentPrice * 0.0002);
        const newPrice = Number((currentPrice + delta).toFixed(symbol.includes('JPY') ? 3 : symbol.includes('BHD') ? 4 : symbol.includes('BTC') ? 2 : 5));

        this.prices.set(symbol, newPrice);
        history.push({ timestamp: now, price: newPrice });

        if (history.length > 300) {
          history.shift();
        }

        const symbolListeners = this.listeners.get(symbol);
        if (symbolListeners) {
          symbolListeners.forEach((cb) => cb(newPrice, history));
        }
      });
    }, 200);
  }

  public getPrice(symbol: string): number {
    return this.prices.get(symbol) || 1.08450;
  }

  public getAllPrices(): Record<string, number> {
    const obj: Record<string, number> = {};
    this.prices.forEach((v, k) => { obj[k] = v; });
    return obj;
  }

  public getHistory(symbol: string): PricePoint[] {
    return this.histories.get(symbol) || [];
  }

  public subscribe(symbol: string, callback: (price: number, history: PricePoint[]) => void) {
    if (!this.listeners.has(symbol)) {
      this.listeners.set(symbol, []);
    }
    this.listeners.get(symbol)!.push(callback);
    return () => {
      const list = this.listeners.get(symbol);
      if (list) {
        this.listeners.set(symbol, list.filter((cb) => cb !== callback));
      }
    };
  }
}

  const hashString = `${trade.id}-${trade.entryTime}-${trade.stake}-${settlementPrice}`;

    hash = (hash << 5) - hash + hashString.charCodeAt(i);
    hash |= 0;
  }

    receiptId: `RCT-${Math.random().toString(36).substring(2, 7).toUpperCase()}-${Date.now().toString().slice(-4)}`,
    tradeId: trade.id,

export function middleware(req: NextRequest) {
  const isAdminRoute = req.nextUrl.pathname.startsWith('/admin');
  const isLoginPage = req.nextUrl.pathname === '/admin/login';
  
  // Check for YOUR specific cookie name
  const session = req.cookies.get('nova_admin_session')?.value;

  // If trying to access admin dashboard without login, redirect to login
  if (isAdminRoute && !isLoginPage && !session) {
    return NextResponse.redirect(new URL('/admin/login', req.url));
  }

  // If already logged in and trying to visit login page, redirect to dashboard
  if (isLoginPage && session) {
    return NextResponse.redirect(new URL('/admin', req.url));
  }

  return NextResponse.next();
}
